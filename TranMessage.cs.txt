using Confluent.Kafka;
using Newtonsoft.Json;
using System;
using System.Threading;
using KafkaConsumer; // Ensure this matches the namespace in TransactionMessage.cs

class Program
{
    static void Main(string[] args)
    {
        var config = new ConsumerConfig
        {
            GroupId = "transaction-consumer-group",
            BootstrapServers = "localhost:9092",
            AutoOffsetReset = AutoOffsetReset.Earliest
        };

        using (var consumer = new ConsumerBuilder<Ignore, string>(config).Build())
        {
            consumer.Subscribe("your-topic-name");

            CancellationTokenSource cts = new CancellationTokenSource();
            Console.CancelKeyPress += (_, e) => {
                e.Cancel = true;
                cts.Cancel();
            };

            try
            {
                while (true)
                {
                    var consumeResult = consumer.Consume(cts.Token);
                    var message = JsonConvert.DeserializeObject<TransactionMessage>(consumeResult.Message.Value);
                    
                    // Process the message
                    Console.WriteLine($"Consumed message with ExternalTicketId: {message.ExternalTicketId}, TransactionType: {message.TransactionType}");
                    
                    // Add your processing logic here
                }
            }
            catch (OperationCanceledException)
            {
                // Ensure the consumer leaves the group cleanly and final offsets are committed.
                consumer.Close();
            }
        }
    }
}
